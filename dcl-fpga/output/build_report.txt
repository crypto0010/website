================================================================================
  1.  PROJECT OVERVIEW
================================================================================

The DCL-FPGA project implements a Discrete Chromatic Labeling accelerator
targeting the Digilent Nexys 4 DDR board.  The design includes:

  Compute Cores:
    - GCD Unit         : Stein's binary GCD for 64-bit numbers (x2 instances)
    - PowerMap Unit    : Binary exponentiation  x^m mod n
    - MulMod Unit      : 64-bit modular multiplication (Russian peasant)
    - Coprime Checker  : Validates edge coprimality using embedded BRAM

  Control Interface:
    - UART 115200-8N1 command protocol (7 commands)
    - Command dispatcher routing to compute cores
    - Demo mode via board switches and buttons

  Memory:
    - Label BRAM  : 32 x 64-bit
    - Edge BRAM   : 256 x (u8, u8) vertex pairs

  I/O:
    - 7-segment multiplexed display (8 digits)
    - 16 LEDs + RGB LED16
    - 16 switches + 5 buttons
    - UART TX/RX

  RTL Source Files (10 modules):
    rtl/dcl_top.v          Top-level integration
    rtl/gcd_unit.v         Stein's binary GCD (64-bit, area-optimized)
    rtl/mulmod_64.v        64-bit modular multiplication
    rtl/power_map_unit.v   Binary exponentiation with modular arithmetic
    rtl/coprime_checker.v  Sequential coprimality checker using BRAM
    rtl/cmd_dispatch.v     UART command parser and dispatcher
    rtl/uart_rx.v          UART receiver (115200 baud, 8N1)
    rtl/uart_tx.v          UART transmitter
    rtl/seven_seg.v        8-digit multiplexed 7-segment display driver
    rtl/demo_ctrl.v        Standalone demo controller

--- rtl/gcd_unit.v (Area-Optimized GCD) ----------------------------------------

`timescale 1ns / 1ps
// Stein's binary GCD -- 64-bit iterative FSM
// Area-optimized: all CTZ and barrel shift operations replaced with
// iterative 1-bit shifts (eliminates priority encoders and barrel shifters).

module gcd_unit (
    input  wire        clk,
    input  wire        rst_n,
    input  wire        start,
    input  wire [63:0] a,
    input  wire [63:0] b,
    output reg  [63:0] result,
    output reg         done,
    output reg         is_coprime
);

    localparam S_IDLE         = 4'd0,
               S_DONE_ZERO    = 4'd1,
               S_STRIP_COMMON = 4'd2,
               S_STRIP_A      = 4'd3,
               S_STRIP_B      = 4'd4,
               S_LOOP_CMP     = 4'd5,
               S_LOOP_SUB     = 4'd6,
               S_FINISH_SHIFT = 4'd7,
               S_FINISH_OUT   = 4'd8;

    reg [3:0]  state;
    reg [63:0] ra, rb;
    reg [6:0]  shift;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state      <= S_IDLE;
            result     <= 64'd0;
            done       <= 1'b0;
            is_coprime <= 1'b0;
            ra         <= 64'd0;
            rb         <= 64'd0;
            shift      <= 7'd0;
        end else begin
            case (state)
                S_IDLE: begin
                    done <= 1'b0;
                    if (start) begin
                        ra    <= a;
                        rb    <= b;
                        shift <= 7'd0;
                        if (a == 64'd0 || b == 64'd0)
                            state <= S_DONE_ZERO;
                        else
                            state <= S_STRIP_COMMON;
                    end
                end

                S_DONE_ZERO: begin
                    result     <= ra | rb;
                    is_coprime <= (ra | rb) == 64'd1;
                    done       <= 1'b1;
                    state      <= S_IDLE;
                end

                S_STRIP_COMMON: begin
                    if (ra[0] | rb[0])
                        state <= S_STRIP_A;
                    else begin
                        ra    <= {1'b0, ra[63:1]};
                        rb    <= {1'b0, rb[63:1]};
                        shift <= shift + 7'd1;
                    end
                end

                S_STRIP_A: begin
                    if (ra[0])
                        state <= S_STRIP_B;
                    else
                        ra <= {1'b0, ra[63:1]};
                end

                S_STRIP_B: begin
                    if (rb[0])
                        state <= S_LOOP_CMP;
                    else
                        rb <= {1'b0, rb[63:1]};
                end

                S_LOOP_CMP: begin
                    if (ra > rb) begin
                        ra <= rb;
                        rb <= ra;
                    end
                    state <= S_LOOP_SUB;
                end

                S_LOOP_SUB: begin
                    rb <= rb - ra;
                    if (rb == ra)
                        state <= S_FINISH_SHIFT;
                    else
                        state <= S_STRIP_B;
                end

                S_FINISH_SHIFT: begin
                    if (shift == 7'd0)
                        state <= S_FINISH_OUT;
                    else begin
                        ra    <= {ra[62:0], 1'b0};
                        shift <= shift - 7'd1;
                    end
                end

                S_FINISH_OUT: begin
                    result     <= ra;
                    is_coprime <= (ra == 64'd1);
                    done       <= 1'b1;
                    state      <= S_IDLE;
                end

                default: state <= S_IDLE;
            endcase
        end
    end
endmodule


--- rtl/mulmod_64.v (Area-Optimized Modular Multiply) --------------------------

`timescale 1ns / 1ps
// 64-bit modular multiplication: result = (a * b) mod m
// Algorithm: Russian peasant binary method
// Area-optimized: saturating mode uses 64-bit + overflow flag instead of 128-bit.
// Modular mode uses iterative restoring division for a mod m.

module mulmod_64 (
    input  wire        clk,
    input  wire        rst_n,
    input  wire        start,
    input  wire [63:0] a,
    input  wire [63:0] b,
    input  wire [63:0] m,
    output reg  [63:0] result,
    output reg         done
);

    localparam S_IDLE      = 3'd0,
               S_REDUCE    = 3'd1,
               S_CALC_INIT = 3'd2,
               S_CALC      = 3'd3,
               S_DONE      = 3'd4;

    reg [2:0]   state;
    reg [63:0]  ra;
    reg [63:0]  rb;
    reg [63:0]  acc;
    reg         saturating;
    reg         saturated;
    reg         ra_overflow;
    reg [64:0]  rem;
    reg [5:0]   bit_cnt;

    wire [64:0] sum_acc_ra = {1'b0, acc} + {1'b0, ra};
    wire [64:0] sum_ra_ra  = {1'b0, ra}  + {1'b0, ra};

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state  <= S_IDLE;
            result <= 0;
            done   <= 0;
            ra <= 0; rb <= 0; acc <= 0;
            saturating  <= 0;
            saturated   <= 0;
            ra_overflow <= 0;
            rem     <= 0;
            bit_cnt <= 0;
        end else begin
            case (state)
                S_IDLE: begin
                    done <= 0;
                    if (start) begin
                        rb  <= b;
                        acc <= 0;
                        if (m == 0) begin
                            saturating  <= 1;
                            saturated   <= 0;
                            ra_overflow <= 0;
                            ra    <= a;
                            state <= S_CALC;
                        end else begin
                            saturating <= 0;
                            saturated  <= 0;
                            ra      <= a;
                            rem     <= 65'd0;
                            bit_cnt <= 6'd63;
                            state   <= S_REDUCE;
                        end
                    end
                end

                S_REDUCE: begin
                    if ({rem[63:0], ra[63]} >= {1'b0, m})
                        rem <= {rem[63:0], ra[63]} - {1'b0, m};
                    else
                        rem <= {rem[63:0], ra[63]};
                    ra <= {ra[62:0], 1'b0};
                    if (bit_cnt == 0)
                        state <= S_CALC_INIT;
                    else
                        bit_cnt <= bit_cnt - 1;
                end

                S_CALC_INIT: begin
                    ra    <= rem[63:0];
                    state <= S_CALC;
                end

                S_CALC: begin
                    if (rb == 0) begin
                        result <= saturated ? 64'hFFFFFFFF_FFFFFFFF : acc;
                        done   <= 1;
                        state  <= S_IDLE;
                    end else begin
                        if (saturating) begin
                            if (!saturated) begin
                                if (rb[0]) begin
                                    if (ra_overflow || sum_acc_ra[64])
                                        saturated <= 1;
                                    else
                                        acc <= sum_acc_ra[63:0];
                                end
                                if (!ra_overflow) begin
                                    if (ra[63])
                                        ra_overflow <= 1;
                                    ra <= {ra[62:0], 1'b0};
                                end
                            end
                        end else begin
                            if (rb[0]) begin
                                acc <= (sum_acc_ra >= {1'b0, m}) ?
                                       sum_acc_ra[63:0] - m : sum_acc_ra[63:0];
                            end
                            ra <= (sum_ra_ra >= {1'b0, m}) ?
                                  sum_ra_ra[63:0] - m : sum_ra_ra[63:0];
                        end
                        rb <= rb >> 1;
                    end
                end

                S_DONE: begin
                    done  <= 1;
                    state <= S_IDLE;
                end

                default: state <= S_IDLE;
            endcase
        end
    end
endmodule


--- rtl/power_map_unit.v -------------------------------------------------------

`timescale 1ns / 1ps
// Binary exponentiation: result = x^m mod modulus
// When modulus == 0: unbounded mode with u64 saturation
// Reuses mulmod_64 for each multiply/square step

module power_map_unit (
    input  wire        clk,
    input  wire        rst_n,
    input  wire        start,
    input  wire [63:0] x,
    input  wire [31:0] m,
    input  wire [63:0] modulus,
    output reg  [63:0] result,
    output reg         done
);

    localparam S_IDLE       = 4'd0,
               S_CHECK_BIT  = 4'd1,
               S_WAIT_MUL   = 4'd2,
               S_MULTIPLY   = 4'd3,
               S_WAIT_SQ    = 4'd4,
               S_SQUARE     = 4'd5,
               S_SHIFT      = 4'd6,
               S_DONE       = 4'd7;

    reg [3:0]   state;
    reg [63:0]  acc;
    reg [63:0]  base;
    reg [31:0]  exp;
    reg         mul_start;
    reg [63:0]  mul_a, mul_b;
    wire [63:0] mul_result;
    wire        mul_done;

    mulmod_64 u_mulmod (
        .clk(clk), .rst_n(rst_n), .start(mul_start),
        .a(mul_a), .b(mul_b), .m(modulus),
        .result(mul_result), .done(mul_done)
    );

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state     <= S_IDLE;
            result    <= 0;
            done      <= 0;
            acc       <= 0;
            base      <= 0;
            exp       <= 0;
            mul_start <= 0;
            mul_a     <= 0;
            mul_b     <= 0;
        end else begin
            mul_start <= 0;

            case (state)
                S_IDLE: begin
                    done <= 0;
                    if (start) begin
                        exp  <= m;
                        acc  <= 1;
                        base <= x;
                        state <= S_CHECK_BIT;
                    end
                end

                S_CHECK_BIT: begin
                    if (exp == 0) begin
                        result <= (acc == 0) ? 64'd1 : acc;
                        done   <= 1;
                        state  <= S_IDLE;
                    end else if (exp[0]) begin
                        mul_a     <= acc;
                        mul_b     <= base;
                        mul_start <= 1;
                        state     <= S_WAIT_MUL;
                    end else begin
                        mul_a     <= base;
                        mul_b     <= base;
                        mul_start <= 1;
                        state     <= S_WAIT_SQ;
                    end
                end

                S_WAIT_MUL: begin
                    if (!mul_done)
                        state <= S_MULTIPLY;
                end

                S_MULTIPLY: begin
                    if (mul_done) begin
                        acc <= mul_result;
                        mul_a     <= base;
                        mul_b     <= base;
                        mul_start <= 1;
                        state     <= S_WAIT_SQ;
                    end
                end

                S_WAIT_SQ: begin
                    if (!mul_done)
                        state <= S_SQUARE;
                end

                S_SQUARE: begin
                    if (mul_done) begin
                        base <= mul_result;
                        state <= S_SHIFT;
                    end
                end

                S_SHIFT: begin
                    exp   <= exp >> 1;
                    state <= S_CHECK_BIT;
                end

                S_DONE: begin
                    done  <= 1;
                    state <= S_IDLE;
                end

                default: state <= S_IDLE;
            endcase
        end
    end
endmodule
